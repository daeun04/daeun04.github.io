예제 8-10 이벤트 함수

핵심: 버튼의 onclick 이벤트로 함수를 호출하고, getElementById().innerHTML로 문단에 글자를 출력합니다.

원본 문제수정
버튼 뒤의 //버튼 클릭 이벤트 — HTML 영역에서는 //가 주석이 아니라서 화면에 글자로 그대로 보임<!-- 버튼 클릭 이벤트 --> HTML 주석으로 변경
<meta charset>, lang 없음 — 서버 설정에 따라 한글이 깨질 수 있음<html lang="ko">, <meta charset="UTF-8"> 추가
<!DOCTYPE html>
<html lang="ko">
<head>
<meta charset="UTF-8">
<title>예제8-10 이벤트함수(Function)</title>
<script>
function jsFunction() {
    alert("눌렀군요!");
    document.getElementById("first").innerHTML = "first";   // 문자 출력
    document.getElementById("second").innerHTML = "second"; // 문자 출력
}
</script>
</head>
<body>
<h1>JavaScript 이벤트함수(Function)</h1>
    <!-- 버튼 클릭 이벤트 -->
    <button onclick="jsFunction()">눌러주세요</button>
    <p id="first"></p>
    <p id="second"></p>
</body>
</html>

예제 8-11 함수 연산

핵심: 입력창의 .value는 문자열이므로 parseInt()로 정수로 바꾼 뒤 더해야 합니다. (안 바꾸면 "1"+"2" = "12")

원본 문제수정
가장 큰 오류: type="text"　id="x" 사이가 전각 공백(U+3000)이라 브라우저가 속성 이름을 "　id"로 인식 → 입력창에 id가 없음 → getElementById("x")가 null → 확인 버튼을 눌러도 동작하지 않음 (입력1, 입력2 둘 다)일반 공백(반각 스페이스)으로 교체
id="sum"size="10" — 속성 사이 공백 없음 (문법 오류)id="sum" size="10"
숫자가 아닌 값을 넣으면 합계에 NaN 표시isNaN()으로 검사 후 안내 (선택 사항)
charset 없음<meta charset="UTF-8"> 추가
<!DOCTYPE html>
<html lang="ko">
<head>
<meta charset="UTF-8">
<title>예제8-11 함수(Function) 계산</title>
<script>
function calc() {
    var x = document.getElementById("x").value; // 입력된 정수1 값
    var y = document.getElementById("y").value; // 입력된 정수2 값
    var tot = parseInt(x) + parseInt(y);        // 입력1,2를 정수로 변환 후 더함
    if (isNaN(tot)) {
        alert("정수를 올바르게 입력하세요.");
        return;
    }
    document.getElementById("sum").value = tot; // id="sum"에 결과 출력
}
</script>
</head>
<body>
<h3>함수 계산 결과</h3>
<h4>두 개의 정수를 입력하세요.</h4>
<form>
    정수 입력1: <input type="text" id="x" size="10"><br>
    정수 입력2: <input type="text" id="y" size="10"><br>
    <input type="button" value="확인" onclick="calc();"><br>
    합-------계: <input type="text" id="sum" size="10" readonly><br>
</form>
</body>
</html>

예제 8-12 내장함수 (prompt, confirm, alert)

함수창 모양반환값
prompt(메시지, 기본값)입력창 + 확인/취소입력한 문자열, 취소 시 null
confirm(메시지)확인/취소확인 true, 취소 false
alert(메시지)확인만없음(undefined)
원본 문제수정
</html> 닫는 태그 누락추가
주석 오타 아니로를, 버튼 이름도 실제로는 '확인/취소''확인'/'취소'를 클릭한 경우로 정정
if(result){ } else {} 가 비어 있어 결과 확인 불가각 경우에 alert로 결과 표시
prompt의 입력값을 받지 않고 버림변수에 저장 후 출력 (취소 시 null 처리)
alert 문구가 "실행할까요?" — alert는 질문에 답할 수 없는 창이라 어색함"경고창입니다!"로 변경
charset 없음, 들여쓰기 탭/공백 혼용정리
<!DOCTYPE html>
<html lang="ko">
<head>
<meta charset="UTF-8">
<title>예제8-12 내장 함수</title>
<script>
function promptMsg() {
    var answer = prompt("무엇이든지 물어보세요?", "질문은 여기에");
    if (answer !== null) {          // 확인을 누른 경우
        alert("입력한 내용: " + answer);
    }
}
function confirmMsg() {
    var result = confirm("계속 진행할까요?");
    if (result) {                   // '확인'을 클릭한 경우 실행
        alert("진행합니다.");
    } else {                        // '취소'를 클릭한 경우 실행
        alert("취소했습니다.");
    }
}
function alertMsg() {
    alert("경고창입니다!");
}
</script>
</head>
<body>
<h3>유용한 함수</h3>
<h4>단추를 누르면 대화상자가 나타납니다.</h4>
<input type="button" value="prompt" onclick="promptMsg()">
<input type="button" value="confirm" onclick="confirmMsg()">
<input type="button" value="alert" onclick="alertMsg()">
</body>
</html>

한눈에 요약

  • 8-10 이벤트(onclick) → 함수 호출 → innerHTML로 출력. HTML 안에서는 // 대신 <!-- --> 주석.
  • 8-11 .value는 문자열 → parseInt()로 변환 후 계산. 속성 사이에는 반드시 일반 공백.
  • 8-12 prompt(문자열/null), confirm(true/false), alert(반환값 없음).
