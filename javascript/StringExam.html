String 활용 예제

문자열 속성과 메서드로 만드는 작은 프로그램 3가지

  • ① 회원가입 검사
  • ② 텍스트 검색 · 변경
  • ③ 성적 데이터 분석
  • 한눈에 정리

이 페이지 사용법

각 프로그램은 왼쪽에서 직접 입력해 볼 수 있고, 오른쪽 실행 과정에 어떤 String 속성과 메서드가 어떤 값을 만들었는지 순서대로 나옵니다. 입력을 바꿀 때마다 오른쪽 값도 바로 바뀌니, 일부러 공백을 넣거나 @를 빼 보면서 각 메서드가 하는 일을 확인해 보세요.

① 회원가입 검사

이름과 이메일을 검사합니다. 공백을 정리하고, 글자 수와 @, 마침표(.)가 있는지 차례로 확인합니다.

trim()toLowerCase()lengthindexOf()includes()
프로그램
실행 과정
    해 볼 것: 이름을 한 글자만 쓰거나, 이메일에서 @나 마침표를 지워 보세요. 처음으로 통과하지 못한 단계에서 검사가 멈추고, 그 뒤 단계는 흐리게 표시됩니다. if ~ else if는 하나라도 걸리면 아래 조건은 보지 않기 때문입니다.
    코드 보기 
    function checkJoin() {
    var name = document.getElementById("name").value;
    var email = document.getElementById("email").value;

    name = name.trim();               // 앞뒤 공백 제거
    email = email.trim();
    email = email.toLowerCase();      // 소문자로 통일

    var nameLength = name.length;     // 이름 글자 수
    var atPosition = email.indexOf("@");  // @ 위치, 없으면 -1
    var hasDot = email.includes(".");     // 마침표 있으면 true

    var message = "";
    if (nameLength == 0) {
        message = "이름을 입력하세요.";
    } else if (nameLength < 2) {
        message = "이름은 2글자 이상 입력하세요.";
    } else if (atPosition == -1) {
        message = "이메일에 @를 입력하세요.";
    } else if (hasDot == false) {
        message = "이메일에 .을 입력하세요.";
    } else {
        message = "가입 정보 확인 완료!\n";
        message += "이름 : " + name + "\n";
        message += "이메일 : " + email;
    }
    document.getElementById("result").innerText = message;
}

    ② 텍스트 검색 · 변경

    문장에서 단어를 찾아 위치와 횟수를 알려 주고, 찾은 단어를 모두 다른 단어로 바꿉니다.

    trim()toLowerCase()lengthincludes()indexOf()split()replaceAll()
    프로그램
    실행 과정
      알아 둘 점: 검색은 문장과 단어를 모두 toLowerCase()로 바꾼 뒤 비교해서 대소문자를 구분하지 않습니다(그래서 "ai"로도 "AI"를 찾습니다). 반면 바꾸기는 replaceAll()을 원래 문장에 그대로 써서 대소문자를 구분합니다. "ai"로는 바꿀 수 없고 "AI"로 입력해야 합니다.
      코드 보기 
      function searchText() {
    var content = document.getElementById("content").value;
    var keyword = document.getElementById("keyword").value.trim();

    var text = content.toLowerCase();     // 대소문자 구분 없이 비교하려고
    var word = keyword.toLowerCase();

    if (word.length == 0) {
        document.getElementById("result").innerText = "검색어를 입력하세요.";
        return;
    }

    var found = text.includes(word);           // 있으면 true
    var position = text.indexOf(word);         // 처음 위치
    var count = text.split(word).length - 1;   // 조각 수 - 1 = 횟수

    if (found) {
        document.getElementById("result").innerText =
            "검색어 : " + keyword + "\n검색 위치 : " + position + "\n검색 횟수 : " + count + "회";
    } else {
        document.getElementById("result").innerText = "검색어를 찾을 수 없습니다.";
    }
}

function replaceText() {
    var content = document.getElementById("content").value;
    var keyword = document.getElementById("keyword").value.trim();
    var newWord = document.getElementById("newWord").value.trim();

    if (keyword.length == 0) { /* "찾을 단어를 입력하세요." */ return; }
    if (newWord.length == 0) { /* "변경할 단어를 입력하세요." */ return; }
    if (content.includes(keyword) == false) { /* "찾을 수 없습니다." */ return; }

    var changedText = content.replaceAll(keyword, newWord);  // 모두 바꾸기
    document.getElementById("changed").innerText = changedText;
    document.getElementById("content").value = changedText;  // 입력창도 바뀐 문장으로
}

      ③ 학생 성적 데이터 분석

      "이름,점수" 형식의 여러 줄을 읽어 학생별 점수, 학생 수, 합계, 평균을 구합니다. 형식이 잘못된 줄은 건너뜁니다.

      split("\n")split(",")trim()lengthNumber()toFixed(2)
      프로그램
      실행 과정 · 한 줄씩 나누고 검사하기
      줄split(",")점수결과
      해 볼 것: 예시에는 일부러 점수가 빈 줄(최지우)과 100점을 넘는 줄(정우성)을 넣었습니다. 둘 다 건너뜀으로 표시되고 평균 계산에서 빠집니다. split("\n")으로 줄을 나누고, 다시 split(",")으로 이름과 점수를 나누는 두 번 자르기가 핵심입니다.
      코드 보기 
      function analyze() {
    var data = document.getElementById("data").value;
    var rows = data.split("\n");          // 줄 단위로 나누기
    var total = 0;
    var validCount = 0;
    var output = "";

    for (var i = 0; i < rows.length; i++) {
        var student = rows[i].split(",");     // ["홍길동", "80"]
        var name = student[0].trim();

        if (student.length < 2 || student[1].trim() == "") {
            continue;                         // 점수가 없으면 건너뜀
        }
        var score = Number(student[1].trim());   // 문자 "80" → 숫자 80
        if (name == "" || isNaN(score) || score < 0 || score > 100) {
            continue;                         // 잘못된 값도 건너뜀
        }
        total = total + score;
        validCount++;
        output += name + " : " + score + "점<br>";
    }

    var average = total / validCount;
    // average.toFixed(2) → 소수점 둘째 자리까지 문자열로
}

      한눈에 정리

      세 프로그램에서 쓴 속성과 메서드입니다.

      속성 · 메서드하는 일예시쓰인 곳
      trim()앞뒤 공백 제거" 홍길동 ".trim() → "홍길동"①②③
      toLowerCase()영문을 모두 소문자로"A@B.COM".toLowerCase() → "a@b.com"①②
      length글자 수 (배열이면 칸 수)"홍길동".length → 3①②③
      indexOf()처음 나오는 위치, 없으면 -1"a@b".indexOf("@") → 1①②
      includes()들어 있으면 true"a.com".includes(".") → true①②
      split()구분자로 잘라 배열로"홍길동,80".split(",") → ["홍길동", "80"]②③
      replaceAll()찾은 단어를 모두 바꿈"a-a".replaceAll("a", "b") → "b-b"②
      Number()문자를 숫자로 바꿈Number("80") → 80③
      toFixed(n)소수점 n자리까지 표시(87.5).toFixed(2) → "87.50"③
      원본 예제와 다른 점: 원본은 세 프로그램이 결과 칸에 모두 같은 id="result"를 써서, 검색 결과와 성적 분석 결과가 회원가입 결과 칸에 출력되는 문제가 있었습니다. 이 페이지는 프로그램마다 다른 id를 써서 각자의 칸에 나오도록 고쳤습니다. 성적 분석의 학생 수도 문자열을 다시 나눠 세는 대신 validCount++로 바로 셉니다.
